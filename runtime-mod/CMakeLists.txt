cmake_minimum_required(VERSION 3.24)
project(Season13Runtime LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(Season13Core STATIC src/PeImage.cpp src/Season13BuildGuard.cpp src/PatternScanner.cpp src/RuntimeLog.cpp src/Memory.cpp)
target_include_directories(Season13Core PUBLIC include)
target_compile_definitions(Season13Core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(Season13Core PUBLIC /W4 /permissive- /EHsc /guard:cf /Zi)
target_link_libraries(Season13Core PUBLIC bcrypt)
add_library(Season13Runtime SHARED src/dllmain.cpp src/Season13Runtime.cpp src/RuntimeAddresses.cpp src/Unreal/ObjectArray.cpp src/Unreal/FName.cpp)
target_link_libraries(Season13Runtime PRIVATE Season13Core)
add_executable(Season13Controller src/controller.cpp)
target_link_libraries(Season13Controller PRIVATE Season13Core)
add_executable(Season13Tests tests/CoreTests.cpp src/Unreal/ObjectArray.cpp)
target_link_libraries(Season13Tests PRIVATE Season13Core)
enable_testing()
add_test(NAME OfflineCore COMMAND Season13Tests)
foreach(t Season13Runtime Season13Controller Season13Tests)
  target_link_options(${t} PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf /DEBUG)
endforeach()
